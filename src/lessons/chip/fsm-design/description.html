A Finite State Machine is the go-to design pattern for any circuit that needs to follow a sequence of steps — traffic lights, USB protocols, CPU controllers, you name it.

Moore vs Mealy

FSM Coding Template

// State encoding
localparam IDLE = 2'b00, START = 2'b01,
           RUN  = 2'b10, DONE  = 2'b11;

reg [1:0] state, next_state;

// State register
always @(posedge clk)
  if (reset) state <= IDLE;
  else       state <= next_state;

// Next-state logic (combinational)
always @(*) begin
  next_state = state; // default: stay
  case (state)
    IDLE:  if (go) next_state = START;
    START: next_state = RUN;
    RUN:   if (done) next_state = DONE;
    DONE:  next_state = IDLE;
  endcase
end

// Output logic (Moore)
assign busy = (state == RUN);
